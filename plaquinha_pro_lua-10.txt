local SAVE_FILE = "plaquinha_pro_save.txt"

local function salvarCfg()
    if not writefile then return end
    local d = cfg.texto1 .."|".. cfg.texto2 .."|"..
              cfg.emoji1esq .."|".. cfg.emoji1dir .."|"..
              cfg.emoji2esq .."|".. cfg.emoji2dir .."|"..
              tostring(cfg.mostrarPlaca1) .."|".. tostring(cfg.mostrarPlaca2) .."|"..
              tostring(cfg.mostrarEmoji1) .."|".. tostring(cfg.mostrarEmoji2) .."|"..
              cfg.largura .."|".. cfg.altura .."|"..
              cfg.posYPlaca1 .."|".. cfg.posYPlaca2 .."|"..
              tostring(cfg.animacao) .."|".. cfg.velocidade .."|".. cfg.ordem
    pcall(writefile, SAVE_FILE, d)
end

local function carregarCfg()
    if not (readfile and isfile and isfile(SAVE_FILE)) then return end
    local ok, d = pcall(readfile, SAVE_FILE)
    if not ok or not d then return end
    local t = d:split("|")
    cfg.texto1         = t[1]  or cfg.texto1
    cfg.texto2         = t[2]  or cfg.texto2
    cfg.emoji1esq      = t[3]  or cfg.emoji1esq
    cfg.emoji1dir      = t[4]  or cfg.emoji1dir
    cfg.emoji2esq      = t[5]  or cfg.emoji2esq
    cfg.emoji2dir      = t[6]  or cfg.emoji2dir
    cfg.mostrarPlaca1  = t[7]  == "true"
    cfg.mostrarPlaca2  = t[8]  == "true"
    cfg.mostrarEmoji1  = t[9]  == "true"
    cfg.mostrarEmoji2  = t[10] == "true"
    cfg.largura        = tonumber(t[11]) or cfg.largura
    cfg.altura         = tonumber(t[12]) or cfg.altura
    cfg.posYPlaca1     = tonumber(t[13]) or cfg.posYPlaca1
    cfg.posYPlaca2     = tonumber(t[14]) or cfg.posYPlaca2
    cfg.animacao       = t[15] == "true"
    cfg.velocidade     = tonumber(t[16]) or cfg.velocidade
    cfg.ordem          = tonumber(t[17]) or cfg.ordem
end

           i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType ~= Enum.UserInputType.MouseMovement and
           i.UserInputType ~= Enum.UserInputType.Touch then return end
        local delta = i.Position - dragStart
        painel.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end)
end

local function mkSlider(sc, label, valor, min, max, passo, callback)
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1,-4,0,54)
    row.BackgroundColor3 = Color3.fromRGB(15,10,30)
    row.BorderSizePixel = 0
    row.ZIndex = 31; row.Parent = sc
    Instance.new("UICorner", row).CornerRadius = UDim.new(0.1,0)

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(0.65,0,0,22)
    lbl.Position = UDim2.new(0,10,0,5)
    lbl.BackgroundTransparency = 1
    lbl.Text = label; lbl.TextScaled = true
    lbl.Font = Enum.Font.GothamBold
    lbl.TextColor3 = Color3.fromRGB(210,200,235)
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.ZIndex = 32; lbl.Parent = row

    local valBox = Instance.new("TextBox")
    valBox.Size = UDim2.new(0.28,0,0,22)
    valBox.Position = UDim2.new(0.7,0,0,5)
    valBox.BackgroundColor3 = Color3.fromRGB(22,15,50)
    valBox.BorderSizePixel = 0
    valBox.Text = tostring(valor); valBox.TextScaled = true
    valBox.Font = Enum.Font.GothamBold
    valBox.TextColor3 = Color3.fromRGB(185,120,255)
    valBox.ClearTextOnFocus = false
    valBox.ZIndex = 32; valBox.Parent = row
    Instance.new("UICorner", valBox).CornerRadius = UDim.new(0.25,0)
    do local s=Instance.new("UIStroke"); s.Color=Color3.fromRGB(90,45,190); s.Thickness=1; s.Parent=valBox end

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1,-16,0,8)
    track.Position = UDim2.new(0,8,1,-16)
    track.BackgroundColor3 = Color3.fromRGB(30,20,55)
    track.BorderSizePixel = 0
    track.ZIndex = 32; track.Parent = row
    Instance.new("UICorner", track).CornerRadius = UDim.new(1,0)

    local pct = math.clamp((valor-min)/(max-min),0,1)
    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(pct,0,1,0)
    fill.BackgroundColor3 = Color3.fromRGB(100,35,210)
    fill.BorderSizePixel = 0
    fill.ZIndex = 33; fill.Parent = track
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1,0)
    do
        local g=Instance.new("UIGradient")
        g.Color=ColorSequence.new({
            ColorSequenceKeypoint.new(0,Color3.fromRGB(60,15,160)),
            ColorSequenceKeypoint.new(1,Color3.fromRGB(200,70,255)),
        }); g.Parent=fill
    end

    local handle = Instance.new("TextButton")
    handle.Size = UDim2.new(0,20,0,20)
    handle.Position = UDim2.new(pct,-10,0.5,-10)
    handle.BackgroundColor3 = Color3.fromRGB(240,230,255)
    handle.BorderSizePixel = 0; handle.Text = ""; handle.ZIndex = 34; handle.Parent = track
    Instance.new("UICorner", handle).CornerRadius = UDim.new(1,0)
    do local s=Instance.new("UIStroke"); s.Color=Color3.fromRGB(140,70,255); s.Thickness=2; s.Parent=handle end

    local function aplicar(np)
        local nv = math.clamp(min + math.round((np*(max-min))/passo)*passo, min, max)
        local np2 = (nv-min)/(max-min)
        fill.Size = UDim2.new(np2,0,1,0)
        handle.Position = UDim2.new(np2,-10,0.5,-10)
        valBox.Text = tostring(nv)
        callback(nv); salvarCfg()
    end

    local dragging = false
    handle.MouseButton1Down:Connect(function() dragging=true end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or
           i.UserInputType==Enum.UserInputType.Touch then dragging=false end
    end)
    UIS.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType~=Enum.UserInputType.MouseMovement and
           i.UserInputType~=Enum.UserInputType.Touch then return end
        local np=math.clamp((i.Position.X-track.AbsolutePosition.X)/track.AbsoluteSize.X,0,1)
        aplicar(np)
    end)
    valBox.FocusLost:Connect(function()
        local n=tonumber(valBox.Text)
        if n then aplicar((math.clamp(n,min,max)-min)/(max-min)) end
    end)
end

atualizarPosAutor = function()
    local bottomY = 0
    if cfg.mostrarPlaca1 and cfg.mostrarPlaca2 then
        local y1 = p1.frame.Position.Y.Offset + cfg.altura
        local y2 = p2.frame.Position.Y.Offset + cfg.altura
        bottomY = math.max(y1, y2)
    elseif cfg.mostrarPlaca1 then
        bottomY = p1.frame.Position.Y.Offset + cfg.altura
    elseif cfg.mostrarPlaca2 then
        bottomY = p2.frame.Position.Y.Offset + cfg.altura
    else
        -- nenhuma placa visível: usa referência da placa1
        bottomY = cfg.posYPlaca1 + cfg.altura
    end
    autorFrame.Position  = UDim2.new(0.5, -(AUTOR_W/2),   0, bottomY + 5)
    autorShadow.Position = UDim2.new(0.5, -(AUTOR_W/2)-2, 0, bottomY + 7)
    -- créditos sempre visíveis
    autorFrame.Visible  = true
    autorShadow.Visible = true
end
